# Build-time tooling for Scripts/make-dmg.sh. Install into a throwaway venv:
#
#   python3 -m venv .dmg-venv
#   ./.dmg-venv/bin/pip install --require-hashes -r Scripts/dmg-requirements.txt
#
# dmgbuild is the DMG layout tool: it drives hdiutil and writes the .DS_Store
# holding icon positions and the background reference via the ds_store and
# mac_alias modules. No AppleScript and no Finder, which is the whole reason
# it is here rather than create-dmg — see Scripts/make-dmg.sh's header.
#
# All three are MIT-licensed and build-time only: nothing from them is
# redistributed inside Cheerio.app, so THIRD-PARTY-NOTICES.md doesn't cover
# them, for the same reason it doesn't cover XcodeGen.
#
# Pinned to exact versions *with hashes* so a release cannot silently pick up
# a different DMG builder. --require-hashes makes pip refuse anything else,
# including a re-uploaded artefact under the same version. To bump: change the
# version, run `pip download --dest /tmp/w <pkg>==<ver>`, and replace the
# sha256 with `shasum -a 256` of the wheel.
dmgbuild==1.6.7 \
    --hash=sha256:37ee5771c377beb3203d9164aae8046ffed8531c06edf9227f5788b3c599b1bf
ds_store==1.3.3 \
